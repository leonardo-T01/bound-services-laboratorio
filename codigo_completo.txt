package com.hfad.odometro_laboratorio;

import android.content.ComponentName;
import android.content.Context;
import android.content.Intent;
import android.content.ServiceConnection;
import android.content.pm.PackageManager;
import android.location.Location;
import android.os.Bundle;
import android.os.Handler;
import android.os.IBinder;
import android.os.Looper;
import android.widget.Button;
import android.widget.EditText;
import android.widget.TextView;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;
import androidx.core.app.ActivityCompat;
import androidx.core.content.ContextCompat;

import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Locale;

public class MainActivity extends AppCompatActivity {

    private static final int PERMISSION_REQUEST_CODE = 698;

    private OdometerService odometer;
    private boolean bound = false;
    private TrackDbHelper db;

    private EditText etEstudiante, etPrecision, etSegundos;
    private TextView tvUbicacion, tvDistancia, tvReporte;

    private final ServiceConnection connection = new ServiceConnection() {
        @Override
        public void onServiceConnected(ComponentName name, IBinder binder) {
            odometer = ((OdometerService.OdometerBinder) binder).getOdometer();
            bound = true;
        }

        @Override
        public void onServiceDisconnected(ComponentName name) {
            bound = false;
        }
    };

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        db = new TrackDbHelper(this);
        etEstudiante = findViewById(R.id.etEstudiante);
        etPrecision = findViewById(R.id.etPrecision);
        etSegundos = findViewById(R.id.etSegundos);
        tvUbicacion = findViewById(R.id.tvUbicacion);
        tvDistancia = findViewById(R.id.tvDistancia);
        tvReporte = findViewById(R.id.tvReporte);

        Button btnIniciar = findViewById(R.id.btnIniciar);
        Button btnDetener = findViewById(R.id.btnDetener);
        Button btnReporte = findViewById(R.id.btnReporte);

        btnIniciar.setOnClickListener(v -> iniciar());
        btnDetener.setOnClickListener(v -> detenerYGuardar());
        btnReporte.setOnClickListener(v -> tvReporte.setText(db.reporte()));

        displayDistance();
    }

    private void iniciar() {
        if (!bound) {
            Toast.makeText(this, "Servicio no enlazado (Â¿permiso de ubicaciÃ³n?)",
                    Toast.LENGTH_SHORT).show();
            return;
        }
        float precision;
        long segundos;
        try {
            precision = Float.parseFloat(etPrecision.getText().toString().trim());
            segundos = Long.parseLong(etSegundos.getText().toString().trim());
        } catch (NumberFormatException e) {
            Toast.makeText(this, "Ingrese precisiÃ³n (m) y tiempo (s) vÃ¡lidos",
                    Toast.LENGTH_SHORT).show();
            return;
        }
        odometer.resetDistance();
        if (!odometer.startTracking(segundos, precision)) {
            Toast.makeText(this, "No se pudo iniciar: revise permiso y GPS",
                    Toast.LENGTH_LONG).show();
        }
    }

    private void detenerYGuardar() {
        if (!bound || !odometer.isTracking()) return;
        odometer.stopTracking();

        Location loc = odometer.getCurrentLocation();
        String estudiante = etEstudiante.getText().toString().trim();
        if (estudiante.isEmpty()) estudiante = "sin_nombre";
        String fecha = new SimpleDateFormat("yyyy-MM-dd HH:mm", Locale.getDefault())
                .format(new Date());

        db.insertar(estudiante, fecha, odometer.getDistance(),
                Float.parseFloat(etPrecision.getText().toString().trim()),
                Long.parseLong(etSegundos.getText().toString().trim()),
                loc != null ? loc.getLatitude() : 0,
                loc != null ? loc.getLongitude() : 0);

        Toast.makeText(this, "Recorrido guardado", Toast.LENGTH_SHORT).show();
        tvReporte.setText(db.reporte());
    }

    /** Actualiza la UI cada segundo con los datos del servicio. */
    private void displayDistance() {
        final Handler handler = new Handler(Looper.getMainLooper());
        handler.post(new Runnable() {
            @Override
            public void run() {
                double distance = 0.0;
                if (bound && odometer != null) {
                    distance = odometer.getDistance();
                    Location loc = odometer.getCurrentLocation();
                    if (loc != null) {
                        tvUbicacion.setText(String.format(Locale.getDefault(),
                                "UbicaciÃ³n actual: %.6f, %.6f",
                                loc.getLatitude(), loc.getLongitude()));
                    }
                }
                tvDistancia.setText(String.format(Locale.getDefault(),
                        "%,.1f metros", distance));
                handler.postDelayed(this, 1000);
            }
        });
    }

    @Override
    protected void onStart() {
        super.onStart();
        if (ContextCompat.checkSelfPermission(this, OdometerService.PERMISSION_STRING)
                != PackageManager.PERMISSION_GRANTED) {
            ActivityCompat.requestPermissions(this,
                    new String[]{OdometerService.PERMISSION_STRING,
                            android.Manifest.permission.ACCESS_COARSE_LOCATION},
                    PERMISSION_REQUEST_CODE);
        } else {
            bindOdometer();
        }
    }

    private void bindOdometer() {
        Intent intent = new Intent(this, OdometerService.class);
        bindService(intent, connection, Context.BIND_AUTO_CREATE);
    }

    @Override
    public void onRequestPermissionsResult(int requestCode, String[] permissions,
                                           int[] grantResults) {
        super.onRequestPermissionsResult(requestCode, permissions, grantResults);
        if (requestCode == PERMISSION_REQUEST_CODE) {
            if (grantResults.length > 0
                    && grantResults[0] == PackageManager.PERMISSION_GRANTED) {
                bindOdometer();
            } else {
                Toast.makeText(this, "Se requiere permiso de ubicaciÃ³n",
                        Toast.LENGTH_LONG).show();
            }
        }
    }

    @Override
    protected void onStop() {
        super.onStop();
        if (bound) {
            unbindService(connection);
            bound = false;
        }
    }
}

package com.hfad.odometro_laboratorio;

import android.app.Service;
import android.content.Context;
import android.content.Intent;
import android.content.pm.PackageManager;
import android.location.Location;
import android.location.LocationListener;
import android.location.LocationManager;
import android.os.Binder;
import android.os.Bundle;
import android.os.IBinder;
import android.os.Looper;

import androidx.core.content.ContextCompat;

/** Bound service: calcula la distancia recorrida (metros) usando el GPS. */
public class OdometerService extends Service {

    public static final String PERMISSION_STRING =
            android.Manifest.permission.ACCESS_FINE_LOCATION;

    private final IBinder binder = new OdometerBinder();
    private LocationManager locManager;
    private LocationListener listener;

    private static double distanceInMeters = 0;
    private static Location lastLocation = null;
    private static Location currentLocation = null;
    private boolean tracking = false;

    public class OdometerBinder extends Binder {
        OdometerService getOdometer() {
            return OdometerService.this;
        }
    }

    @Override
    public IBinder onBind(Intent intent) {
        return binder;
    }

    @Override
    public void onCreate() {
        super.onCreate();
        locManager = (LocationManager) getSystemService(Context.LOCATION_SERVICE);
        listener = new LocationListener() {
            @Override
            public void onLocationChanged(Location location) {
                if (lastLocation != null) {
                    distanceInMeters += location.distanceTo(lastLocation);
                }
                lastLocation = location;
                currentLocation = location;
            }

            @Override public void onProviderDisabled(String provider) {}
            @Override public void onProviderEnabled(String provider) {}
            @Override public void onStatusChanged(String provider, int status, Bundle extras) {}
        };
    }

    /**
     * Inicia las actualizaciones de ubicaciÃ³n fijando el GPS_PROVIDER explÃ­citamente.
     */
    public boolean startTracking(long seconds, float meters) {
        if (ContextCompat.checkSelfPermission(this, PERMISSION_STRING)
                != PackageManager.PERMISSION_GRANTED) {
            return false;
        }
        stopTracking();

        // Forzar el proveedor GPS del emulador en lugar de getBestProvider
        String provider = LocationManager.GPS_PROVIDER;
        if (locManager == null || !locManager.isProviderEnabled(provider)) {
            provider = LocationManager.NETWORK_PROVIDER;
        }

        // Obtener la posiciÃ³n actual de partida
        Location last = locManager.getLastKnownLocation(provider);
        if (last != null) {
            currentLocation = last;
            lastLocation = last;
        }

        try {
            // Escuchar actualizaciones garantizando el Looper principal
            locManager.requestLocationUpdates(
                    provider,
                    seconds * 1000L,
                    meters,
                    listener,
                    Looper.getMainLooper()
            );
            tracking = true;
            return true;
        } catch (Exception e) {
            e.printStackTrace();
            return false;
        }
    }

    public void stopTracking() {
        if (tracking && locManager != null && listener != null
                && ContextCompat.checkSelfPermission(this, PERMISSION_STRING)
                == PackageManager.PERMISSION_GRANTED) {
            locManager.removeUpdates(listener);
        }
        tracking = false;
    }

    public void resetDistance() {
        distanceInMeters = 0;
        lastLocation = currentLocation;
    }

    public boolean isTracking() { return tracking; }

    public double getDistance() { return distanceInMeters; }

    public Location getCurrentLocation() { return currentLocation; }

    @Override
    public void onDestroy() {
        super.onDestroy();
        stopTracking();
        locManager = null;
        listener = null;
    }
}

package com.hfad.odometro_laboratorio;

import android.content.ContentValues;
import android.content.Context;
import android.database.Cursor;
import android.database.sqlite.SQLiteDatabase;
import android.database.sqlite.SQLiteOpenHelper;

/** Persistencia (SQLite) de los recorridos para el reporte consolidado. */
public class TrackDbHelper extends SQLiteOpenHelper {

    private static final String DB_NAME = "odometro.db";
    private static final int DB_VERSION = 1;

    public TrackDbHelper(Context context) {
        super(context, DB_NAME, null, DB_VERSION);
    }

    @Override
    public void onCreate(SQLiteDatabase db) {
        db.execSQL("CREATE TABLE recorrido (" +
                "id INTEGER PRIMARY KEY AUTOINCREMENT," +
                "estudiante TEXT," +
                "fecha TEXT," +
                "distancia_m REAL," +
                "precision_m REAL," +
                "intervalo_s INTEGER," +
                "latitud REAL," +
                "longitud REAL)");
    }

    @Override
    public void onUpgrade(SQLiteDatabase db, int oldV, int newV) {
        db.execSQL("DROP TABLE IF EXISTS recorrido");
        onCreate(db);
    }

    public void insertar(String estudiante, String fecha, double distancia,
                         float precision, long intervalo, double lat, double lon) {
        ContentValues v = new ContentValues();
        v.put("estudiante", estudiante);
        v.put("fecha", fecha);
        v.put("distancia_m", distancia);
        v.put("precision_m", precision);
        v.put("intervalo_s", intervalo);
        v.put("latitud", lat);
        v.put("longitud", lon);
        getWritableDatabase().insert("recorrido", null, v);
    }

    /** Reporte consolidado: lista de recorridos + total por estudiante. */
    public String reporte() {
        StringBuilder sb = new StringBuilder();
        SQLiteDatabase db = getReadableDatabase();

        Cursor c = db.rawQuery("SELECT estudiante, fecha, distancia_m, precision_m, intervalo_s " +
                "FROM recorrido ORDER BY id DESC", null);
        while (c.moveToNext()) {
            sb.append(String.format("%s | %s | %.1f m (prec %.0f m, c/%d s)\n",
                    c.getString(0), c.getString(1), c.getDouble(2),
                    c.getDouble(3), c.getLong(4)));
        }
        c.close();

        sb.append("\n--- TOTAL POR ESTUDIANTE ---\n");
        Cursor t = db.rawQuery("SELECT estudiante, COUNT(*), SUM(distancia_m) " +
                "FROM recorrido GROUP BY estudiante", null);
        while (t.moveToNext()) {
            sb.append(String.format("%s: %d recorridos, %.1f m en total\n",
                    t.getString(0), t.getInt(1), t.getDouble(2)));
        }
        t.close();
        return sb.toString();
    }
}
